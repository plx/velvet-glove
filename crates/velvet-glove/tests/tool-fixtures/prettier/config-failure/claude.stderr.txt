def resolve_program(requested, label, must_be_executable):
    resolved = requested if os.path.dirname(requested) else shutil.which(requested)
    if resolved is None:
        fail(f"cannot resolve {label} {requested!r}")
    resolved = os.path.realpath(resolved)
    try:
        info = os.stat(resolved)
    except OSError as error:
        fail(f"cannot inspect {label} {resolved}: {error}")
    if not stat.S_ISREG(info.st_mode) or not os.access(resolved, os.R_OK):
        fail(f"resolved {label} is not a readable regular file: {resolved}")
    if must_be_executable and not os.access(resolved, os.X_OK):
        fail(f"resolved {label} is not executable: {resolved}")
    return resolved
