def write_owned_file(path, data):
    descriptor = None
    try:
        descriptor = os.open(
            path,
            os.O_WRONLY
            | os.O_CREAT
            | os.O_EXCL
            | getattr(os, "O_NOFOLLOW", 0),
            0o600,
        )
        os.fchmod(descriptor, 0o600)
        cursor = 0
        while cursor < len(data):
            written = os.write(descriptor, data[cursor:])
            if written <= 0:
                abort(f"short write while creating controlled input {path}")
            cursor += written
        os.fsync(descriptor)
        info = os.fstat(descriptor)
        if (
            not stat.S_ISREG(info.st_mode)
            or info.st_nlink != 1
            or stat.S_IMODE(info.st_mode) != 0o600
            or info.st_size != len(data)
        ):
            abort(f"controlled input has unsafe metadata: {path}")
    except OSError as error:
        abort(f"cannot create controlled input {path}: {error}")
    finally:
        if descriptor is not None:
            try:
                os.close(descriptor)
            except OSError:
                pass

def group_alive(process_group):
    try:
        os.killpg(process_group, 0)
        return True
    except ProcessLookupError:
        return False
    except (OSError, ValueError):
        return True


def signal_group(process_group, signum):
    try:
        os.killpg(process_group, signum)
    except (OSError, ValueError):
        pass


def sweep_group(process_group, initial_signal=signal.SIGTERM):
    signal_group(process_group, initial_signal)
    deadline = time.monotonic() + 1.0
    while group_alive(process_group) and time.monotonic() < deadline:
        time.sleep(0.01)
    if group_alive(process_group):
        signal_group(process_group, signal.SIGKILL)
        deadline = time.monotonic() + 1.0
        while group_alive(process_group) and time.monotonic() < deadline:
            time.sleep(0.01)
    if group_alive(process_group):
        raise RuntimeError("child process group survived SIGKILL")
