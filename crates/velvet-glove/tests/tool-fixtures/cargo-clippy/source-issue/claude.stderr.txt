def file_snapshot(path):
    initial = os.lstat(path)
    if stat.S_ISLNK(initial.st_mode) or not stat.S_ISREG(initial.st_mode) or initial.st_nlink != 1:
        raise ValueError(f"path is not a unique regular file: {path}")
    flags = os.O_RDONLY | getattr(os, "O_NOFOLLOW", 0)
    descriptor = os.open(path, flags)
    with os.fdopen(descriptor, "rb", closefd=True) as source_file:
        opened = os.fstat(source_file.fileno())
        if (
            not stat.S_ISREG(opened.st_mode)
            or opened.st_nlink != 1
            or opened.st_dev != initial.st_dev
            or opened.st_ino != initial.st_ino
        ):
            raise ValueError(f"path changed while it was opened: {path}")
        content = source_file.read(LIMIT + 1)
        final = os.fstat(source_file.fileno())
    if len(content) > LIMIT:
        raise ValueError(f"path exceeds the 16 MiB validation limit: {path}")
    stable_fields = ("st_dev", "st_ino", "st_mode", "st_nlink", "st_size", "st_mtime_ns")
    if any(getattr(opened, field) != getattr(final, field) for field in stable_fields):
        raise ValueError(f"path changed while it was read: {path}")
    return {
        "device": final.st_dev,
        "inode": final.st_ino,
        "mode": stat.S_IMODE(final.st_mode),
        "size": final.st_size,
        "mtime": final.st_mtime_ns,
        "sha256": hashlib.sha256(content).hexdigest(),
        "content": content,
    }

class AdapterSignal(BaseException):
    def __init__(self, signum):
        super().__init__(f"received signal {signum}")
        self.signum = signum
def signal_group(signum):
    if process is None:
        return
    try:
        os.killpg(process.pid, signum)
    except (OSError, ValueError):
        pass
def stop_child(initial_signal=None):
    global process
    global cleaning
    child = process
    if child is None:
        return
    cleaning = True
    try:
        try:
            running = child.poll() is None
        except (OSError, ValueError):
            running = True
        if running:
            signal_group(initial_signal if initial_signal is not None else signal.SIGTERM)
            try:
                child.wait(timeout=1)
            except (OSError, ValueError, subprocess.TimeoutExpired):
                pass
        signal_group(signal.SIGKILL)
        try:
            child.wait(timeout=1)
        except (OSError, ValueError, subprocess.TimeoutExpired):
            pass
    finally:
        process = None
        cleaning = False
def forward_signal(signum, _frame):
    global pending_signal
    pending_signal = signum
    if process is not None and not cleaning:
        signal_group(signum)
        raise AdapterSignal(signum)
def run_child(arguments):
    global process
    stdout_chunks = []
    stderr_chunks = []
    total = 0
    process = subprocess.Popen(
        arguments,
        stdin=subprocess.DEVNULL,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        cwd=invocation_dir,
        env=environment,
        start_new_session=True,
    )
    if pending_signal is not None:
        raise AdapterSignal(pending_signal)
    selector = selectors.DefaultSelector()
    for stream, target in ((process.stdout, stdout_chunks), (process.stderr, stderr_chunks)):
        if stream is None:
            raise OSError("failed to capture Cargo output")
        os.set_blocking(stream.fileno(), False)
        selector.register(stream, selectors.EVENT_READ, target)
    while selector.get_map():
        if pending_signal is not None:
            raise AdapterSignal(pending_signal)
        for key, _ in selector.select(timeout=0.1):
            chunk = os.read(key.fileobj.fileno(), 65536)
            if chunk:
                total += len(chunk)
                if total > LIMIT:
                    raise BufferError(f"combined output exceeded {LIMIT} bytes")
                key.data.append(chunk)
            else:
                selector.unregister(key.fileobj)
                key.fileobj.close()
    returncode = process.wait()
    process = None
    if pending_signal is not None:
        raise AdapterSignal(pending_signal)
    return returncode, b"".join(stdout_chunks), b"".join(stderr_chunks)

def iter_diagnostic_nodes(message):
    pending = [message]
    while pending:
        node = pending.pop(0)
        yield node
        children = node.get("children", [])
        if not isinstance(children, list):
            raise ValueError("diagnostic children is not an array")
        for child in children:
            if not isinstance(child, dict):
                raise ValueError("diagnostic child is not an object")
            pending.append(child)
def parse_clippy(raw, returncode, package_id, metadata_targets):
    try:
        text = raw.decode("utf-8")
    except UnicodeError as error:
        raise ValueError(f"Cargo JSON output is not UTF-8: {error}")
    if not text or not text.endswith("\n"):
        raise ValueError("Cargo JSON output is incomplete")
    records = []
    for line_number, line in enumerate(text.splitlines(), 1):
        if line_number > 100000:
            raise ValueError("Cargo JSON contains more than 100000 records")
        try:
            record = json.loads(line)
        except json.JSONDecodeError as error:
            raise ValueError(f"Cargo JSON line {line_number} is invalid: {error}")
        if not isinstance(record, dict) or not isinstance(record.get("reason"), str):
            raise ValueError(f"Cargo JSON line {line_number} has an invalid shape")
        if record["reason"] not in {
            "build-finished",
            "build-script-executed",
            "compiler-artifact",
            "compiler-message",
        }:
            raise ValueError(f"Cargo JSON line {line_number} has an unsupported reason")
        records.append(record)
    finished = [record for record in records if record["reason"] == "build-finished"]
    if len(finished) != 1 or records[-1] is not finished[0] or type(finished[0].get("success")) is not bool:
        raise ValueError("Cargo JSON has no unique terminal build-finished record")
    if (returncode == 0) != finished[0]["success"] or returncode not in (0, 101):
        raise ValueError("Cargo status disagrees with build-finished completion")
    artifact_targets = set()
    artifact_filenames = set()
    diagnostics = {}
    diagnostic_edits = {}
    foreign_error = False
    for record in records:
        reason = record["reason"]
        if reason == "compiler-artifact" and record.get("package_id") == package_id:
            target = record.get("target")
            filenames = record.get("filenames")
            if not isinstance(target, dict):
                raise ValueError("workspace compiler artifact has no target")
            if (
                not isinstance(filenames, list)
                or not filenames
                or any(not isinstance(filename, str) or not filename for filename in filenames)
            ):
                raise ValueError("workspace compiler artifact has invalid output filenames")
            name = target.get("name")
            source = target.get("src_path")
            kinds = target.get("kind")
            if not isinstance(name, str) or not isinstance(source, str) or not isinstance(kinds, list):
                raise ValueError("workspace compiler artifact target is malformed")
            relative = normalized_span_path(source)
            key = (name, relative, tuple(kinds))
            if key not in metadata_targets:
                raise ValueError("workspace compiler artifact is absent from Cargo metadata")
            artifact_targets.add(key)
            for filename in filenames:
                artifact_path = os.path.abspath(filename)
                try:
                    inside_target = os.path.commonpath((target_dir, artifact_path)) == target_dir
                except ValueError:
                    inside_target = False
                if not inside_target or os.path.realpath(artifact_path) != artifact_path:
                    raise ValueError("workspace compiler artifact output escapes the controlled target")
                artifact_filenames.add(os.fsencode(artifact_path))
        elif reason == "compiler-message":
            message = record.get("message")
            if not isinstance(message, dict):
                raise ValueError("compiler-message has no diagnostic object")
            level = message.get("level")
            if level not in ("error", "warning"):
                continue
            if record.get("package_id") != package_id:
                if level == "error":
                    foreign_error = True
                continue
            code_record = message.get("code")
            code = code_record.get("code") if isinstance(code_record, dict) else None
            spans = message.get("spans")
            message_text = message.get("message")

                "file": primary_paths[0],

                "fixable": bool(edits) or (existing is not None and existing["fixable"]),

def makefile_words(raw):
    words = []
    current = bytearray()
    escaped = False
    for byte in raw:
        if escaped:
            if byte == 10:
                escaped = False
                continue
            current.append(byte)
            escaped = False
        elif byte == 92:
            escaped = True
        elif byte in (9, 10, 13, 32):
            if current:
                words.append(bytes(current))
                current.clear()
        else:
            current.append(byte)
    if escaped:
        raise ValueError("Cargo dep-info ends with an incomplete escape")
    if current:
        words.append(bytes(current))
    return words

def same_identity(path, before):
    try:
        current = file_snapshot(path)

    return all(
        current[key] == before[key]

def write_temporary(path, content, mode):
    descriptor, temporary_path = tempfile.mkstemp(prefix=".velvet-glove-clippy-", dir=os.path.dirname(path))
    try:
        os.fchmod(descriptor, mode)
        with os.fdopen(descriptor, "wb", closefd=True) as output:
            descriptor = -1
            output.write(content)
            output.flush()
            os.fsync(output.fileno())
        return temporary_path
    except BaseException:
        if descriptor >= 0:
            os.close(descriptor)
        try:
            os.unlink(temporary_path)
        except OSError:
            pass
        raise
def apply_suggestions(diagnostic_edits):
    edits_by_file = {}
    for edits in diagnostic_edits.values():
        for relative, start, end, replacement in edits:
            edits_by_file.setdefault(relative, set()).add((start, end, replacement))
    transformed = {}
    for relative, unique_edits in edits_by_file.items():
        before = source_snapshots[relative]
        ordered = sorted(unique_edits)
        previous = None
        for edit in ordered:
            start, end, replacement = edit
            if end > len(before["content"]):
                raise ValueError(f"suggestion is outside source bytes: {relative}")
            if previous is not None and (
                start < previous[1]
                or (start == previous[0] and end == previous[1] and replacement != previous[2])
            ):
                raise ValueError(f"machine-applicable suggestions overlap: {relative}")
            previous = edit
        content = before["content"]
        for start, end, replacement in reversed(ordered):
            content = content[:start] + replacement.encode("utf-8") + content[end:]
        if content != before["content"]:
            transformed[relative] = content
    if any(edits for edits in diagnostic_edits.values()) and not transformed:
        raise ValueError("machine-applicable suggestions made no byte changes")
    prepared = {}
    replaced = []
    try:
        for relative, content in transformed.items():
            path = os.path.join(workspace, relative.replace("/", os.sep))
            before = source_snapshots[relative]
            if not same_identity(path, before):
                raise ValueError(f"source changed before applying suggestions: {relative}")
            prepared[relative] = write_temporary(path, content, before["mode"])
        for relative in sorted(prepared):
            path = os.path.join(workspace, relative.replace("/", os.sep))
            before = source_snapshots[relative]
            if not same_identity(path, before):
                raise ValueError(f"source changed during suggestion transaction: {relative}")
            os.replace(prepared.pop(relative), path)
            replaced.append(relative)
    except BaseException as error:
        rollback_errors = []

            path = os.path.join(workspace, relative.replace("/", os.sep))

        raise

            try:
                os.unlink(temporary_path)
            except OSError:
                pass

def file_snapshot(path):
    initial = os.lstat(path)
    if stat.S_ISLNK(initial.st_mode) or not stat.S_ISREG(initial.st_mode) or initial.st_nlink != 1:
        raise ValueError(f"path is not a unique regular file: {path}")
    flags = os.O_RDONLY | getattr(os, "O_NOFOLLOW", 0)
    descriptor = os.open(path, flags)
    with os.fdopen(descriptor, "rb", closefd=True) as source_file:
        opened = os.fstat(source_file.fileno())
        if (
            not stat.S_ISREG(opened.st_mode)
            or opened.st_nlink != 1
            or opened.st_dev != initial.st_dev
            or opened.st_ino != initial.st_ino
        ):
            raise ValueError(f"path changed while it was opened: {path}")
        content = source_file.read(LIMIT + 1)
        final = os.fstat(source_file.fileno())
    if len(content) > LIMIT:
        raise ValueError(f"path exceeds the 16 MiB validation limit: {path}")
    stable_fields = ("st_dev", "st_ino", "st_mode", "st_nlink", "st_size", "st_mtime_ns")
    if any(getattr(opened, field) != getattr(final, field) for field in stable_fields):
        raise ValueError(f"path changed while it was read: {path}")
    return {
        "device": final.st_dev,
        "inode": final.st_ino,
        "mode": stat.S_IMODE(final.st_mode),
        "size": final.st_size,
        "mtime": final.st_mtime_ns,
        "sha256": hashlib.sha256(content).hexdigest(),
        "content": content,
    }

class AdapterSignal(BaseException):
    def __init__(self, signum):
        super().__init__(f"received signal {signum}")
        self.signum = signum
def signal_group(signum):
    if process is None:
        return
    try:
        os.killpg(process.pid, signum)
    except (OSError, ValueError):
        pass
def stop_child(initial_signal=None):
    global process
    global cleaning
    child = process
    if child is None:
        return
    cleaning = True
    try:
        try:
            running = child.poll() is None
        except (OSError, ValueError):
            running = True
        if running:
            signal_group(initial_signal if initial_signal is not None else signal.SIGTERM)
            try:
                child.wait(timeout=1)
            except (OSError, ValueError, subprocess.TimeoutExpired):
                pass
        signal_group(signal.SIGKILL)
        try:
            child.wait(timeout=1)
        except (OSError, ValueError, subprocess.TimeoutExpired):
            pass
    finally:
        process = None
        cleaning = False
def forward_signal(signum, _frame):
    global pending_signal
    pending_signal = signum
    if process is not None and not cleaning:
        signal_group(signum)
        raise AdapterSignal(signum)
def run_child(arguments):
    global process
    stdout_chunks = []
    stderr_chunks = []
    total = 0
    process = subprocess.Popen(
        arguments,
        stdin=subprocess.DEVNULL,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        cwd=invocation_dir,
        env=environment,
        start_new_session=True,
    )
    if pending_signal is not None:
        raise AdapterSignal(pending_signal)
    selector = selectors.DefaultSelector()
    for stream, target in ((process.stdout, stdout_chunks), (process.stderr, stderr_chunks)):
        if stream is None:
            raise OSError("failed to capture Cargo output")
        os.set_blocking(stream.fileno(), False)
        selector.register(stream, selectors.EVENT_READ, target)
    while selector.get_map():
        if pending_signal is not None:
            raise AdapterSignal(pending_signal)
        for key, _ in selector.select(timeout=0.1):
            chunk = os.read(key.fileobj.fileno(), 65536)
            if chunk:
                total += len(chunk)
                if total > LIMIT:
                    raise BufferError(f"combined output exceeded {LIMIT} bytes")
                key.data.append(chunk)
            else:
                selector.unregister(key.fileobj)
                key.fileobj.close()
    returncode = process.wait()
    process = None
    if pending_signal is not None:
        raise AdapterSignal(pending_signal)
    return returncode, b"".join(stdout_chunks), b"".join(stderr_chunks)

def parse_metadata(raw):
    try:
        metadata = json.loads(raw.decode("utf-8"))
    except (UnicodeError, json.JSONDecodeError) as error:
        raise ValueError(f"Cargo metadata is not one UTF-8 JSON object: {error}")
    if not isinstance(metadata, dict):
        raise ValueError("Cargo metadata root is not an object")
    if metadata.get("version") != 1:
        raise ValueError("Cargo metadata does not use the locked version-1 schema")
    metadata_root = metadata.get("workspace_root")
    if not isinstance(metadata_root, str) or os.path.realpath(metadata_root) != workspace:
        raise ValueError("Cargo workspace root differs from the selected Cargo.toml directory")
    target_directory = metadata.get("target_directory")
    if not isinstance(target_directory, str) or os.path.realpath(target_directory) != target_dir:
        raise ValueError("Cargo metadata ignored the controlled target directory")
    build_directory = metadata.get("build_directory")
    if not isinstance(build_directory, str) or os.path.realpath(build_directory) != target_dir:
        raise ValueError("Cargo metadata ignored the controlled build directory")
    members = metadata.get("workspace_members")
    packages = metadata.get("packages")
    if not isinstance(members, list) or len(members) != 1 or not isinstance(members[0], str):
        raise ValueError("the controlled contract requires exactly one workspace package")
    if not isinstance(packages, list) or len(packages) != 1:
        raise ValueError("Cargo metadata must describe exactly one package")
    workspace_packages = [package for package in packages if package.get("id") == members[0]]
    if len(workspace_packages) != 1:
        raise ValueError("Cargo metadata does not identify one workspace package")
    package = workspace_packages[0]
    package_manifest = package.get("manifest_path")
    if not isinstance(package_manifest, str) or os.path.realpath(package_manifest) != manifest:
        raise ValueError("workspace package manifest differs from the selected Cargo.toml")
    targets = package.get("targets")
    if not isinstance(targets, list) or not targets:
        raise ValueError("workspace package has no Cargo targets")
    target_keys = set()
    for target in targets:
        if not isinstance(target, dict):
            raise ValueError("Cargo target is not an object")
        name = target.get("name")
        source = target.get("src_path")
        kinds = target.get("kind")
        if not isinstance(name, str) or not isinstance(source, str) or not isinstance(kinds, list):
            raise ValueError("Cargo target has an invalid shape")
        source_path = os.path.abspath(source)
        try:
            inside = os.path.commonpath((workspace, source_path)) == workspace
        except ValueError:
            inside = False
        if (
            not inside
            or os.path.realpath(source_path) != source_path
            or not source_path.endswith(".rs")
            or "custom-build" in kinds
        ):
            raise ValueError(f"Cargo target is outside the supported Rust source scope: {source}")
        relative = os.path.relpath(source_path, workspace).replace(os.sep, "/")
        if relative not in source_snapshots:
            raise ValueError(f"Cargo target is not a validated physical Rust file: {relative}")
        key = (name, relative, tuple(kinds))
        if key in target_keys:
            raise ValueError("Cargo metadata repeats a target")
        target_keys.add(key)
    package_name = package.get("name")
    if not isinstance(package_name, str) or not package_name:
        raise ValueError("workspace package has no Cargo package name")
    return members[0], package_name, target_keys
def validate_issue_stderr(raw, package_name):
    try:
        text = raw.decode("utf-8")
    except UnicodeError as error:
        raise ValueError(f"Cargo issue summary is not UTF-8: {error}")
    if not text or not text.endswith("\n"):
        raise ValueError("Cargo issue summary is absent or incomplete")
    pattern = re.compile(
        r"^error: could not compile `"
        + re.escape(package_name)
        + r"`(?: \([^\r\n]+\))? due to [1-9][0-9]* previous errors?$"
    )
    lines = text.splitlines()
    if not lines or any(pattern.fullmatch(line) is None for line in lines):
        raise ValueError("Cargo issue summary contains unexpected operational output")
def normalized_span_path(file_name):
    if not isinstance(file_name, str) or not file_name:
        raise ValueError("diagnostic span has no file name")
    candidate = file_name if os.path.isabs(file_name) else os.path.join(workspace, file_name)
    candidate = os.path.abspath(candidate)
    try:
        inside = os.path.commonpath((workspace, candidate)) == workspace
    except ValueError:
        inside = False
    if not inside or os.path.realpath(candidate) != candidate or not candidate.endswith(".rs"):
        raise ValueError(f"diagnostic span escapes the Rust workspace: {file_name}")
    relative = os.path.relpath(candidate, workspace).replace(os.sep, "/")
    if relative not in source_snapshots:
        raise ValueError(f"diagnostic span is not a validated Rust source: {relative}")
    return relative
def iter_diagnostic_nodes(message):
    pending = [message]
    while pending:
        node = pending.pop(0)
        yield node
        children = node.get("children", [])
        if not isinstance(children, list):
            raise ValueError("diagnostic children is not an array")
        for child in children:
            if not isinstance(child, dict):
                raise ValueError("diagnostic child is not an object")
            pending.append(child)
def parse_clippy(raw, returncode, package_id, metadata_targets):
    try:
        text = raw.decode("utf-8")
    except UnicodeError as error:
        raise ValueError(f"Cargo JSON output is not UTF-8: {error}")
    if not text or not text.endswith("\n"):
        raise ValueError("Cargo JSON output is incomplete")
    records = []
    for line_number, line in enumerate(text.splitlines(), 1):
        if line_number > 100000:
            raise ValueError("Cargo JSON contains more than 100000 records")
        try:
            record = json.loads(line)
        except json.JSONDecodeError as error:
            raise ValueError(f"Cargo JSON line {line_number} is invalid: {error}")
        if not isinstance(record, dict) or not isinstance(record.get("reason"), str):
            raise ValueError(f"Cargo JSON line {line_number} has an invalid shape")
        if record["reason"] not in {
            "build-finished",
            "build-script-executed",
            "compiler-artifact",
            "compiler-message",
        }:
            raise ValueError(f"Cargo JSON line {line_number} has an unsupported reason")
        records.append(record)
    finished = [record for record in records if record["reason"] == "build-finished"]
    if len(finished) != 1 or records[-1] is not finished[0] or type(finished[0].get("success")) is not bool:
        raise ValueError("Cargo JSON has no unique terminal build-finished record")
    if (returncode == 0) != finished[0]["success"] or returncode not in (0, 101):
        raise ValueError("Cargo status disagrees with build-finished completion")
    artifact_targets = set()
    artifact_filenames = set()
    diagnostics = {}
    diagnostic_edits = {}
    foreign_error = False
    for record in records:
        reason = record["reason"]
        if reason == "compiler-artifact" and record.get("package_id") == package_id:
            target = record.get("target")
            filenames = record.get("filenames")
            if not isinstance(target, dict):
                raise ValueError("workspace compiler artifact has no target")
            if (
                not isinstance(filenames, list)
                or not filenames
                or any(not isinstance(filename, str) or not filename for filename in filenames)
            ):
                raise ValueError("workspace compiler artifact has invalid output filenames")
            name = target.get("name")
            source = target.get("src_path")
            kinds = target.get("kind")
            if not isinstance(name, str) or not isinstance(source, str) or not isinstance(kinds, list):
                raise ValueError("workspace compiler artifact target is malformed")
            relative = normalized_span_path(source)
            key = (name, relative, tuple(kinds))
            if key not in metadata_targets:
                raise ValueError("workspace compiler artifact is absent from Cargo metadata")
            artifact_targets.add(key)
            for filename in filenames:
                artifact_path = os.path.abspath(filename)
                try:
                    inside_target = os.path.commonpath((target_dir, artifact_path)) == target_dir
                except ValueError:
                    inside_target = False
                if not inside_target or os.path.realpath(artifact_path) != artifact_path:
                    raise ValueError("workspace compiler artifact output escapes the controlled target")
                artifact_filenames.add(os.fsencode(artifact_path))
        elif reason == "compiler-message":
            message = record.get("message")
            if not isinstance(message, dict):
                raise ValueError("compiler-message has no diagnostic object")
            level = message.get("level")
            if level not in ("error", "warning"):
                continue
            if record.get("package_id") != package_id:
                if level == "error":
                    foreign_error = True
                continue
            code_record = message.get("code")
            code = code_record.get("code") if isinstance(code_record, dict) else None
            spans = message.get("spans")
            message_text = message.get("message")
            if not isinstance(code, str) or not code or not isinstance(spans, list) or not isinstance(message_text, str):
                foreign_error = True
                continue
            primary = [span for span in spans if isinstance(span, dict) and span.get("is_primary") is True]
            if not primary:
                foreign_error = True
                continue
            try:
                primary_paths = [normalized_span_path(span.get("file_name")) for span in primary]
            except ValueError:
                foreign_error = True
                continue
            first = primary[0]
            line = first.get("line_start")
            column = first.get("column_start")
            if type(line) is not int or line < 1 or type(column) is not int or column < 1:
                raise ValueError("workspace diagnostic has invalid source coordinates")
            diagnostic_key = (code, primary_paths[0], line, column, message_text)
            edits = set()
            for node in iter_diagnostic_nodes(message):
                node_spans = node.get("spans", [])
                if not isinstance(node_spans, list):
                    raise ValueError("diagnostic spans is not an array")
                replacements = [
                    span
                    for span in node_spans
                    if isinstance(span, dict) and span.get("suggested_replacement") is not None
                ]
                if not replacements:
                    continue
                if any(
                    span.get("suggestion_applicability") != "MachineApplicable"
                    or not isinstance(span.get("suggested_replacement"), str)
                    for span in replacements
                ):
                    continue
                for span in replacements:
                    relative = normalized_span_path(span.get("file_name"))
                    start = span.get("byte_start")
                    end = span.get("byte_end")
                    replacement = span.get("suggested_replacement")
                    if type(start) is not int or type(end) is not int or start < 0 or end < start:
                        raise ValueError("machine-applicable suggestion has invalid byte offsets")
                    edits.add((relative, start, end, replacement))
            existing = diagnostics.get(diagnostic_key)
            diagnostics[diagnostic_key] = {
                "code": code,
                "file": primary_paths[0],
                "line": line,
                "column": column,
                "message": message_text,
                "fixable": bool(edits) or (existing is not None and existing["fixable"]),
            }
            diagnostic_edits.setdefault(diagnostic_key, set()).update(edits)
    if foreign_error:
        raise ValueError("Cargo reported a non-source or dependency compilation error")
    if returncode == 0:
        if diagnostics or artifact_targets != metadata_targets:
            raise ValueError("clean Cargo completion lacks exact target coverage or has diagnostics")
        return [], {}, "clean", artifact_filenames
    if not diagnostics:
        raise ValueError("failed Cargo completion has no validated workspace source diagnostic")
    return (
        [diagnostics[key] for key in sorted(diagnostics)],
        diagnostic_edits,
        "issues",
        artifact_filenames,
    )
def makefile_words(raw):
    words = []
    current = bytearray()
    escaped = False
    for byte in raw:
        if escaped:
            if byte == 10:
                escaped = False
                continue
            current.append(byte)
            escaped = False
        elif byte == 92:
            escaped = True
        elif byte in (9, 10, 13, 32):
            if current:
                words.append(bytes(current))
                current.clear()
        else:
            current.append(byte)
    if escaped:
        raise ValueError("Cargo dep-info ends with an incomplete escape")
    if current:
        words.append(bytes(current))
    return words
def validate_source_coverage(require_complete, selected_artifact_filenames):
    covered = set()
    dep_info_count = 0
    selected_dep_info_count = 0
    dep_info_bytes = 0
    def walk_error(error):
        raise ValueError(f"cannot walk controlled Cargo target directory: {error}")
    for directory, child_directories, names in os.walk(
        target_dir, followlinks=False, onerror=walk_error
    ):
        for name in child_directories:
            path = os.path.join(directory, name)
            if os.path.islink(path):
                raise ValueError(f"Cargo target directory contains a symlink directory: {path}")
        for name in names:
            if not name.endswith(".d"):
                continue
            path = os.path.join(directory, name)
            try:
                info = os.lstat(path)
                if os.path.islink(path) or not stat.S_ISREG(info.st_mode) or info.st_nlink != 1:
                    raise ValueError(f"Cargo dep-info is not a unique regular file: {path}")
                with open(path, "rb") as dep_info_file:
                    raw = dep_info_file.read(16 * 1024 * 1024 + 1)
            except OSError as error:
                raise ValueError(f"cannot read Cargo dep-info {path}: {error}")
            if len(raw) > 16 * 1024 * 1024:
                raise ValueError(f"Cargo dep-info exceeds the 16 MiB limit: {path}")
            dep_info_count += 1
            if dep_info_count > 8192:
                raise ValueError("Cargo produced more than 8192 dep-info files")
            dep_info_bytes += len(raw)
            if dep_info_bytes > 64 * 1024 * 1024:
                raise ValueError("Cargo dep-info exceeds the 64 MiB aggregate limit")
            words = makefile_words(raw)
            rule_targets = {word[:-1] for word in words if word.endswith(b":")}
            if rule_targets.isdisjoint(selected_artifact_filenames):
                continue
            selected_dep_info_count += 1
            for word in words:
                if word.endswith(b":") or not word.endswith(b".rs"):
                    continue
                try:
                    source = os.fsdecode(word)
                except UnicodeError as error:
                    raise ValueError(f"Cargo dep-info path is not valid text: {error}")
                candidate = source if os.path.isabs(source) else os.path.join(workspace, source)
                candidate = os.path.abspath(candidate)
                try:
                    inside = os.path.commonpath((workspace, candidate)) == workspace
                except ValueError:
                    inside = False
                if not inside:
                    continue
                if os.path.realpath(candidate) != candidate:
                    raise ValueError(f"Cargo dep-info source is not canonical: {source}")
                relative = os.path.relpath(candidate, workspace).replace(os.sep, "/")
                if relative not in source_snapshots:
                    raise ValueError(f"Cargo compiled an unvalidated workspace source: {relative}")
                covered.add(relative)
    if dep_info_count == 0 or selected_dep_info_count == 0:
        raise ValueError("Cargo produced no selected-package source dependency evidence")
    if require_complete and covered != set(source_snapshots):
        missing = sorted(set(source_snapshots) - covered)
        raise ValueError(
            "Cargo did not compile every physical workspace Rust source: " + ", ".join(missing)
        )
    return len(covered)
def same_identity(path, before):
    try:
        current = file_snapshot(path)
    except (OSError, ValueError):
        return False
    return all(
        current[key] == before[key]
        for key in ("device", "inode", "mode", "size", "mtime", "sha256")
    )
def write_temporary(path, content, mode):
    descriptor, temporary_path = tempfile.mkstemp(prefix=".velvet-glove-clippy-", dir=os.path.dirname(path))
    try:
        os.fchmod(descriptor, mode)
        with os.fdopen(descriptor, "wb", closefd=True) as output:
            descriptor = -1
            output.write(content)
            output.flush()
            os.fsync(output.fileno())
        return temporary_path
    except BaseException:
        if descriptor >= 0:
            os.close(descriptor)
        try:
            os.unlink(temporary_path)
        except OSError:
            pass
        raise
def apply_suggestions(diagnostic_edits):
    edits_by_file = {}
    for edits in diagnostic_edits.values():
        for relative, start, end, replacement in edits:
            edits_by_file.setdefault(relative, set()).add((start, end, replacement))
    transformed = {}
    for relative, unique_edits in edits_by_file.items():
        before = source_snapshots[relative]
        ordered = sorted(unique_edits)
        previous = None
        for edit in ordered:
            start, end, replacement = edit
            if end > len(before["content"]):
                raise ValueError(f"suggestion is outside source bytes: {relative}")
            if previous is not None and (
                start < previous[1]
                or (start == previous[0] and end == previous[1] and replacement != previous[2])
            ):
                raise ValueError(f"machine-applicable suggestions overlap: {relative}")
            previous = edit
        content = before["content"]
        for start, end, replacement in reversed(ordered):
            content = content[:start] + replacement.encode("utf-8") + content[end:]
        if content != before["content"]:
            transformed[relative] = content
    if any(edits for edits in diagnostic_edits.values()) and not transformed:
        raise ValueError("machine-applicable suggestions made no byte changes")
    prepared = {}
    replaced = []
    try:
        for relative, content in transformed.items():
            path = os.path.join(workspace, relative.replace("/", os.sep))
            before = source_snapshots[relative]
            if not same_identity(path, before):
                raise ValueError(f"source changed before applying suggestions: {relative}")
            prepared[relative] = write_temporary(path, content, before["mode"])
        for relative in sorted(prepared):
            path = os.path.join(workspace, relative.replace("/", os.sep))
            before = source_snapshots[relative]
            if not same_identity(path, before):
                raise ValueError(f"source changed during suggestion transaction: {relative}")
            os.replace(prepared.pop(relative), path)
            replaced.append(relative)
    except BaseException as error:
        rollback_errors = []
        for relative in reversed(replaced):
            path = os.path.join(workspace, relative.replace("/", os.sep))
            before = source_snapshots[relative]
            try:
                restore = write_temporary(path, before["content"], before["mode"])
                os.replace(restore, path)
            except BaseException as rollback_error:
                rollback_errors.append(f"{relative}: {rollback_error}")
        if rollback_errors:
            raise ValueError(f"{error}; rollback also failed: {'; '.join(rollback_errors)}")
        raise
    finally:
        for temporary_path in prepared.values():
            try:
                os.unlink(temporary_path)
            except OSError:
                pass
    return sorted(transformed)
